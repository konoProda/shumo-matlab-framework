% 说明：必须用 XeLaTeX 编译（ctex 需要），PDFLaTeX 会在中文字体处报错。
% 详细编译与打包说明见同目录 README.md。

\documentclass[12pt,a4paper]{ctexart}

% ---- 版面 ----
\usepackage[a4paper,top=2.7cm,bottom=2.7cm,left=2.6cm,right=2.6cm]{geometry}

% ---- 数学 ----
\usepackage{amsmath,amssymb,bm}

% ---- 表格与图 ----
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{graphicx}
\usepackage{float}
\usepackage{caption}

% ---- 列表 ----
\usepackage{enumitem}

% ---- 代码（附录 B）----
\usepackage{fvextra}

% ---- 横版页（题目强制表 1/2/3 需横版）----
% pdflscape 会把该页旋转 90°，阅读时无需歪头；若发行版缺该宏包，
% 换成 lscape 也能编译，只是页面不自动旋转。
\usepackage{pdflscape}

% ---- 行距 ----
\usepackage{setspace}

% ---- 标题 ----
\usepackage{titlesec}

% ============================================================
% 编号设置
% ============================================================
% 章节号必须显式重定义：titlesec 的 \titleformat 只改**显示标签**，
% 不改 \thesection 本身，若不重定义，交叉引用会得到阿拉伯数字，
% 出现"一、…"与"1.1"对不上的错号（2025C 骨架里就是靠 \numberwithin 硬凑的）。
\renewcommand{\thesection}{\chinese{section}}
\renewcommand{\thesubsection}{\arabic{section}.\arabic{subsection}}
\renewcommand{\thesubsubsection}%
  {\arabic{section}.\arabic{subsection}.\arabic{subsubsection}}

% 公式编号**全局连续**。
% 严禁 \numberwithin{equation}{section}：本文正文共 85 个编号公式，
% 一旦按节编号就会变成 5.1 … 8.31，与初稿的 tag 1..85 全部错位。
\renewcommand{\theequation}{\arabic{equation}}

\renewcommand{\figurename}{图}
\renewcommand{\tablename}{表}
\renewcommand{\refname}{参考文献}

% ============================================================
% 标题格式
% ============================================================
\titleformat{\section}{\centering\heiti\zihao{3}}{\thesection、}{0pt}{}
\titlespacing*{\section}{0pt}{0.8\baselineskip}{0.5\baselineskip}

\titleformat{\subsection}{\heiti\zihao{4}}{\thesubsection}{0.8em}{}
\titlespacing*{\subsection}{0pt}{0.5\baselineskip}{0.3\baselineskip}

\titleformat{\subsubsection}{\heiti\zihao{-4}}{\thesubsubsection}{0.6em}{}
\titlespacing*{\subsubsection}{0pt}{0.4\baselineskip}{0.2\baselineskip}

% ============================================================
% 版式细节
% ============================================================
\graphicspath{{figures/}}

\captionsetup{
  font=small,
  labelfont=bf,
  labelsep=quad,
  skip=6pt
}

\setlist[itemize]{leftmargin=2em}
\setlist[enumerate]{leftmargin=2em}

% 列间距收窄：默认 6pt 下，5 列以上的表会顶出版心（实测 q3 的指定日期表
% 达 7 列，超版心约 45%）。收到 4pt 后，除该表外全部落回版心内。
\setlength{\tabcolsep}{4pt}

% 行距：1.28 时全文 50 页，压到单倍是页数达标的主要手段（每页多出约 6 行）
\setstretch{1.0}
\setlength{\parindent}{2em}
\setlength{\parskip}{0pt}
% 公式上下间距收到 3pt：全文行间公式 160 余处，累计可省约 2 页
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\setlength{\floatsep}{8pt plus 2pt minus 2pt}
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}
\setlength{\intextsep}{8pt plus 2pt minus 2pt}

\pagestyle{plain}

% 附录 B 的代码：朴素无高亮、无行号、无边框。
% breaklines/breakanywhere 把超宽行**排版时软折行**——源码本身一字不改，
% 附录与支撑材料保持同一份文本，便于评审核对（源码由 ../src/ 复制而来）。
\fvset{
  fontsize=\fontsize{7.5pt}{9pt}\selectfont,   % 约六号，为把附录代码压到 ~30 页
  breaklines=true,
  breakanywhere=true,
  breaksymbolleft={},
  breaksymbolright={},
  numbers=none,
  frame=none,
  tabsize=2
}
